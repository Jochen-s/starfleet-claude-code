# Secret Detection Patterns for Red Alert -- Phase A2
# One regex per line. Comments start with #.
# Used with: grep -rn -E -f patterns/secrets.txt {target}
# Exclude: node_modules .git __pycache__ venv .venv dist build
# Verify matches are not in test/mock/example files before confirming Critical.

# --- Generic API keys and tokens ---
(?i)(api[_-]?key|api[_-]?secret|access[_-]?token|auth[_-]?token|bearer)\s*[:=]\s*['"][a-zA-Z0-9_\-]{16,}['"]

# --- AWS credentials ---
AKIA[0-9A-Z]{16}
(?i)(aws[_-]?secret[_-]?access[_-]?key)\s*[:=]\s*['"][a-zA-Z0-9/+]{40}['"]

# --- Private key headers (PEM format) ---
-----BEGIN (RSA |EC |DSA |OPENSSH )?PRIVATE KEY-----

# --- Password and secret assignments ---
(?i)(password|passwd|pwd|secret)\s*[:=]\s*['"][^'"]{8,}['"]

# --- Database and service connection strings with embedded credentials ---
(?i)(mongodb|postgres|postgresql|mysql|redis|amqp|rabbitmq)://[^:]+:[^@]{3,}@

# --- Static JWT tokens (three base64url segments) ---
eyJ[a-zA-Z0-9_-]{10,}\.[a-zA-Z0-9_-]{10,}\.[a-zA-Z0-9_-]{10,}

# --- Anthropic API keys ---
sk-ant-[a-zA-Z0-9]{20,}

# --- OpenAI API keys (also matches Anthropic sk-ant- keys; deduplicate at finding level) ---
sk-[a-zA-Z0-9]{48,}

# --- Slack tokens ---
xox[bpors]-[a-zA-Z0-9-]{10,}

# --- GitHub tokens (classic and fine-grained) ---
gh[pousr]_[a-zA-Z0-9]{36,}

# --- Generic hex secrets (32+ char hex assigned to secret/token/key) ---
(?i)(secret|token|key)\s*[:=]\s*['"][0-9a-f]{32,}['"]

# --- Google API keys ---
AIza[0-9A-Za-z_-]{35}

# --- Stripe keys ---
[sr]k_(test|live)_[a-zA-Z0-9]{20,}

# --- SendGrid API keys ---
SG\.[a-zA-Z0-9_-]{22}\.[a-zA-Z0-9_-]{43}

# --- Mailgun API keys (require mailgun context to avoid false positives) ---
(?i)mailgun.*key-[0-9a-zA-Z]{32}

# --- Generic bearer token assignment in config/source ---
(?i)bearer\s+[a-zA-Z0-9_\-\.]{20,}

# --- Hardcoded basic auth in URLs ---
https?://[a-zA-Z0-9_\-]+:[a-zA-Z0-9_\-!@#]{6,}@
